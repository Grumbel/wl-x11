# By default wl-x11 builds the patched tree under subprojects/wlroots.
# System wlroots lacks MODE-before-map, size-only WM_NORMAL_HINTS, and
# wlr_x11_output_get_window() — enabling this will almost certainly break
# placement and property syncing unless those patches are reapplied.
option(
  'use_system_wlroots',
  type: 'boolean',
  value: false,
  description: 'Link against pkg-config wlroots instead of the vendored patched subproject (not recommended)',
)

# Packaging (e.g. Nix flake) may set this to VERSION + "+g" + short git rev.
# Empty means use the contents of the top-level VERSION file as-is.
option(
  'version_full',
  type: 'string',
  value: '',
  description: 'Override full version string for WLX_VERSION (default: VERSION file)',
)
